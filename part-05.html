Praktik E-Report / Part 05

Daftar, login, dan logout

Pengunjung dapat membuat akun warga dan masuk.

Langkah praktik

php artisan make:livewire Auth/Register --class --no-interaction
php artisan make:livewire Auth/Login --class --no-interaction
php artisan make:controller LogoutController --no-interaction

Setiap komponen punya dua file: PHP untuk proses dan Blade untuk tampilan. Ganti isi pasangan file di bawah. wire:model menghubungkan input ke variabel PHP; wire:submit memanggil method saat formulir dikirim.

register() memeriksa isian, menyimpan akun sebagai warga, lalu masuk otomatis. Password di-hash oleh model User. login() memeriksa kredensial. Logout mengakhiri sesi. Pengunjung tidak diberi pilihan mendaftar sebagai admin.

Validasi artinya memeriksa isian sebelum disimpan. Contohnya email wajib unik, nomor WhatsApp 9–15 digit, dan password warga minimal 8 karakter berisi huruf serta angka.

Kode lengkap

Buka satu file, salin seluruh isinya, lalu simpan ke jalur yang tertera di dalam project latihan. Untuk file yang sudah ada, ganti isinya. File Blade dan konfigurasi boleh dibuat lewat VS Code. Pelajari satu pasangan PHP–Blade dalam satu sesi.

app/Livewire/Auth/Register.php
<?php

namespace App\Livewire\Auth;

use App\Models\User;
use Illuminate\Contracts\View\View;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Str;
use Illuminate\Validation\Rules\Password;
use Illuminate\Validation\ValidationException;
use Livewire\Attributes\Layout;
use Livewire\Component;

#[Layout('layouts.auth')]
class Register extends Component
{
    public string $name = '';

    public string $email = '';

    public string $no_whatsapp = '';

    public string $alamat = '';

    public string $password = '';

    public string $password_confirmation = '';

    public function register(): void
    {
        $key = 'register:'.request()->ip();
        if (RateLimiter::tooManyAttempts($key, 10)) {
            throw ValidationException::withMessages(['email' => 'Terlalu banyak pendaftaran. Silakan coba lagi nanti.']);
        }
        $this->email = Str::lower(trim($this->email));
        $data = $this->validate([
            'name' => 'required|string|max:100',
            'email' => 'required|email|max:255|unique:users,email',
            'no_whatsapp' => ['required', 'regex:/^\+?[0-9]{9,15}$/'],
            'alamat' => 'required|string|max:2000',
            'password' => ['required', 'string', 'max:255', 'confirmed', Password::min(8)->letters()->numbers()],
        ]);
        RateLimiter::hit($key, 3600);
        $user = new User($data);
        $user->role = 'warga';
        $user->save();
        Auth::login($user);
        session()->regenerate();
        $this->redirectRoute('warga.dashboard');
    }

    public function render(): View
    {
        return view('livewire.auth.register')->title('Daftar Warga');
    }
}

app/Livewire/Auth/Login.php
<?php

namespace App\Livewire\Auth;

use Illuminate\Contracts\View\View;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;
use Livewire\Attributes\Layout;
use Livewire\Component;

#[Layout('layouts.auth')]
class Login extends Component
{
    public string $email = '';

    public string $password = '';

    public bool $remember = false;

    public function login(): void
    {
        $this->validate(['email' => 'required|email|max:255', 'password' => 'required|string|max:255']);
        $key = 'login:'.Str::transliterate(Str::lower($this->email)).'|'.request()->ip();
        if (RateLimiter::tooManyAttempts($key, 5)) {
            throw ValidationException::withMessages(['email' => 'Terlalu banyak percobaan. Coba lagi dalam '.RateLimiter::availableIn($key).' detik.']);
        }
        if (! Auth::attempt(['email' => Str::lower(trim($this->email)), 'password' => $this->password], $this->remember)) {
            RateLimiter::hit($key, 60);
            $this->reset('password');
            throw ValidationException::withMessages(['email' => 'Email atau password tidak sesuai.']);
        }
        RateLimiter::clear($key);
        session()->regenerate();
        $this->redirectRoute(auth()->user()->dashboardRoute());
    }

    public function render(): View
    {
        return view('livewire.auth.login')->title('Masuk');
    }
}

app/Http/Controllers/LogoutController.php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class LogoutController extends Controller
{
    public function __invoke(Request $request): RedirectResponse
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();

        return redirect()->route('home');
    }
}
