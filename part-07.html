Praktik E-Report / Part 07

Riwayat, detail, edit, dan hapus

Warga dapat mengelola laporan miliknya sendiri.

Langkah praktik

php artisan make:class Services/ComplaintManager --no-interaction
php artisan make:livewire Warga/Laporan/Index --class --no-interaction
php artisan make:livewire Warga/Laporan/Show --class --no-interaction
php artisan make:livewire Warga/Laporan/Edit --class --no-interaction

Index menampilkan riwayat; Show menampilkan detail; Edit mengubah laporan. Cari judul dan pilih status untuk mempersempit daftar. Pagination membagi data menjadi beberapa halaman.

Warga hanya boleh edit atau hapus laporan sendiri yang masih pending. ComplaintManager memeriksa ulang izin ketika menyimpan perubahan. Transaksi menjaga satu rangkaian perubahan database tetap utuh; lockForUpdate mencegah dua perubahan status saling bertabrakan.

Foto lama dibersihkan setelah perubahan berhasil. Tombol hapus memakai konfirmasi supaya pengguna tidak salah klik.

Kode lengkap

Buka satu file, salin seluruh isinya, lalu simpan ke jalur yang tertera di dalam project latihan. Untuk file yang sudah ada, ganti isinya. File Blade dan konfigurasi boleh dibuat lewat VS Code. Pelajari satu pasangan PHP–Blade dalam satu sesi.

app/Services/ComplaintManager.php
<?php

namespace App\Services;

use App\Models\Laporan;
use App\Support\StoredImage;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Gate;
use Illuminate\Validation\ValidationException;

class ComplaintManager
{
    public function update(int $id, array $data): Laporan
    {
        $oldPhoto = null;
        try {
            $laporan = DB::transaction(function () use ($id, $data, &$oldPhoto): Laporan {
                $laporan = Laporan::query()->lockForUpdate()->findOrFail($id);
                Gate::authorize('update', $laporan);
                $oldPhoto = $laporan->foto;
                $laporan->update($data);

                return $laporan;
            });
        } catch (\Throwable $exception) {
            if (isset($data['foto'])) {
                StoredImage::delete($data['foto'], 'laporan');
            }
            throw $exception;
        }
        if (isset($data['foto']) && $oldPhoto !== $data['foto']) {
            StoredImage::delete($oldPhoto, 'laporan');
        }

        return $laporan;
    }

    public function delete(int $id): void
    {
        $photo = DB::transaction(function () use ($id): ?string {
            $laporan = Laporan::query()->lockForUpdate()->findOrFail($id);
            Gate::authorize('delete', $laporan);
            $photo = $laporan->foto;
            $laporan->delete();

            return $photo;
        });
        StoredImage::delete($photo, 'laporan');
    }

    public function transition(int $id, string $status): void
    {
        DB::transaction(function () use ($id, $status): void {
            $laporan = Laporan::query()->lockForUpdate()->findOrFail($id);
            Gate::authorize('changeStatus', $laporan);
            $next = match ($laporan->status) {
                'pending' => 'proses',
                'proses' => 'selesai',
                default => null,
            };
            if ($next === null || $status !== $next) {
                throw ValidationException::withMessages(['status' => 'Perubahan status tidak diizinkan. Ikuti urutan penanganan pengaduan.']);
            }
            $laporan->status = $status;
            $laporan->save();
        });
    }
}

<?php

namespace App\Livewire\Warga\Laporan;

use App\Livewire\WargaPage;
use App\Models\Laporan;
use App\Services\ComplaintManager;
use Illuminate\Contracts\View\View;
use Illuminate\Validation\Rule;
use Livewire\WithPagination;

class Index extends WargaPage
{
    use WithPagination;

    public string $search = '';

    public string $status = '';

    public function updatedSearch(): void
    {
        $this->resetPage();
    }

    public function updatedStatus(): void
    {
        $this->resetPage();
    }

    public function delete(int $id): void
    {
        app(ComplaintManager::class)->delete($id);
        session()->flash('success', 'Pengaduan berhasil dihapus.');
    }

    public function render(): View
    {
        $this->validate(['search' => 'max:255', 'status' => ['nullable', Rule::in(array_keys(Laporan::STATUSES))]]);
        $laporans = auth()->user()->laporans()
            ->when($this->search !== '', fn ($q) => $q->where('judul', 'like', '%'.$this->search.'%'))
            ->when($this->status !== '', fn ($q) => $q->where('status', $this->status))
            ->latest()->orderByDesc('id')->paginate(10);

        return view('livewire.warga.laporan.index', compact('laporans'))->title('Pengaduan Saya');
    }
}

<?php

namespace App\Livewire\Warga\Laporan;

use App\Livewire\ComplaintForm;
use App\Models\Laporan;
use App\Services\ComplaintManager;
use Illuminate\Contracts\View\View;
use Livewire\Attributes\Locked;

class Edit extends ComplaintForm
{
    #[Locked]
    public int $laporanId;

    public function mount(Laporan $laporan): void
    {
        $this->authorize('update', $laporan);
        $this->laporanId = $laporan->id;
        $this->judul = $laporan->judul;
        $this->detail = $laporan->detail;
        $this->tanggal = $laporan->tanggal->format('Y-m-d');
    }

    public function save(): void
    {
        $this->authorize('update', Laporan::findOrFail($this->laporanId));
        app(ComplaintManager::class)->update($this->laporanId, $this->validatedData());
        session()->flash('success', 'Perubahan pengaduan berhasil disimpan.');
        $this->redirectRoute('warga.laporan.show', ['laporan' => $this->laporanId]);
    }

    public function render(): View
    {
        $laporan = Laporan::findOrFail($this->laporanId);
        $this->authorize('update', $laporan);

        return view('livewire.warga.laporan.edit', compact('laporan'))->title('Edit Pengaduan');
    }
}
