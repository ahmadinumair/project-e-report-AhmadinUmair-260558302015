Praktik E-Report / Part 08

Dashboard warga dan profil

Warga bisa melihat ringkasan dan memperbarui identitas.

Langkah praktik

php artisan make:livewire Warga/Dashboard --class --no-interaction
php artisan make:livewire Warga/Profile --class --no-interaction

Dashboard menghitung laporan milik pengguna yang sedang login. Profil mengubah nama, kontak, alamat, foto, dan password. Untuk mengganti password, pengguna harus mengisi password saat ini.

wire:poll mengambil pembaruan secara berkala ketika halaman terbuka. Angka statistik berasal dari database, bukan angka yang ditulis manual di HTML.

Kode lengkap

Buka satu file, salin seluruh isinya, lalu simpan ke jalur yang tertera di dalam project latihan. Untuk file yang sudah ada, ganti isinya. File Blade dan konfigurasi boleh dibuat lewat VS Code. Pelajari satu pasangan PHP–Blade dalam satu sesi.

app/Livewire/Warga/Dashboard.php
<?php

namespace App\Livewire\Warga;

use App\Livewire\WargaPage;
use App\Models\Laporan;
use Illuminate\Contracts\View\View;

class Dashboard extends WargaPage
{
    public function render(): View
    {
        $query = auth()->user()->laporans();
        $counts = (clone $query)->selectRaw('status, count(*) as jumlah')->groupBy('status')->pluck('jumlah', 'status');
        $stats = array_map(fn ($status) => (int) ($counts[$status] ?? 0), array_combine(array_keys(Laporan::STATUSES), array_keys(Laporan::STATUSES)));
        $total = array_sum($stats);
        $recent = $query->latest()->orderByDesc('id')->limit(5)->get();

        return view('livewire.warga.dashboard', compact('stats', 'total', 'recent'))->title('Dashboard Warga');
    }
}

<?php

namespace App\Livewire\Warga;

use App\Livewire\WargaPage;
use App\Models\User;
use App\Support\StoredImage;
use Illuminate\Contracts\View\View;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Illuminate\Validation\Rules\Password;
use Livewire\WithFileUploads;

class Profile extends WargaPage
{
    use WithFileUploads;

    public string $name = '';

    public string $no_whatsapp = '';

    public string $alamat = '';

    public $foto;

    public string $current_password = '';

    public string $password = '';

    public string $password_confirmation = '';

    public function mount(): void
    {
        $user = auth()->user();
        $this->name = $user->name;
        $this->no_whatsapp = $user->no_whatsapp;
        $this->alamat = $user->alamat ?? '';
    }

    public function updateProfile(): void
    {
        $data = $this->validate([
            'name' => 'required|string|max:100',
            'no_whatsapp' => ['required', 'regex:/^\+?[0-9]{9,15}$/'],
            'alamat' => 'required|string|max:2000',
            'foto' => 'nullable|image|mimes:jpg,jpeg,png,webp|max:2048',
        ]);
        unset($data['foto']);
        if ($this->foto) {
            $data['foto'] = $this->foto->store('profil', 'public');
        }
        try {
            $oldPhoto = DB::transaction(function () use ($data): ?string {
                $user = User::query()->lockForUpdate()->findOrFail(auth()->id());
                $oldPhoto = $user->foto;
                $user->update($data);

                return $oldPhoto;
            });
        } catch (\Throwable $exception) {
            StoredImage::delete($data['foto'] ?? null, 'profil');
            throw $exception;
        }
        if (isset($data['foto'])) {
            StoredImage::delete($oldPhoto, 'profil');
        }
        auth()->user()->refresh();
        $this->reset('foto');
        session()->flash('success', 'Profil berhasil diperbarui.');
    }

    public function updatePassword(): void
    {
        $this->validate([
            'current_password' => 'required|current_password',
            'password' => ['required', 'string', 'max:255', 'confirmed', 'different:current_password', Password::min(8)->letters()->numbers()],
        ]);
        $user = auth()->user();
        $user->password = $this->password;
        $user->setRememberToken(Str::random(60));
        $user->save();
        $this->reset('current_password', 'password', 'password_confirmation');
        session()->regenerate();
        session()->flash('success', 'Password berhasil diubah.');
    }

    public function render(): View
    {
        return view('livewire.warga.profile')->title('Profil Saya');
    }
}
