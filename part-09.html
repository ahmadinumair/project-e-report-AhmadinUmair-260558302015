Praktik E-Report / Part 09

Admin, route, dan menjalankan aplikasi

Warga dan admin dapat memakai aplikasi dari awal sampai akhir.

Langkah praktik

php artisan make:livewire Admin/Dashboard --class --no-interaction
php artisan make:livewire Admin/Laporan/Index --class --no-interaction
php artisan make:livewire Admin/Laporan/Show --class --no-interaction
php artisan make:seeder AdminSeeder --no-interaction

Admin melihat seluruh laporan dan mengubah status secara berurutan: pending → proses → selesai. Status tidak boleh dilompati atau dikembalikan. Isi file admin, konfigurasi, bootstrap, dan route di bawah. Route adalah alamat yang menghubungkan URL ke halaman.

Tambahkan ADMIN_EMAIL dan ADMIN_PASSWORD ke .env. Tentukan email admin yang belum dipakai. Buat password sendiri, minimal 12 karakter dengan huruf besar, huruf kecil, angka, dan simbol. Tulis password di dalam tanda kutip ganda.

ADMIN_EMAIL=admin@ereport.test
ADMIN_PASSWORD="GANTI_DENGAN_PASSWORD_BUATANMU"

Teks GANTI_DENGAN_PASSWORD_BUATANMU adalah penanda, bukan password siap pakai. Jalankan:

php artisan config:clear --no-interaction
php artisan db:seed --class=AdminSeeder --no-interaction
php artisan route:list --no-interaction
php artisan serve --no-interaction

Buka localhost:8000/register, buat akun warga, lalu kirim laporan percobaan. Logout dan login memakai akun admin. Buka laporan tadi lalu ubah menjadi proses dan selesai. Seeder tidak mengganti akun yang emailnya sudah ada.

Kode lengkap

Buka satu file, salin seluruh isinya, lalu simpan ke jalur yang tertera di dalam project latihan. Untuk file yang sudah ada, ganti isinya. File Blade dan konfigurasi boleh dibuat lewat VS Code. Pelajari satu pasangan PHP–Blade dalam satu sesi.

app/Livewire/Admin/Dashboard.php
<?php

namespace App\Livewire\Admin;

use App\Livewire\AdminPage;
use App\Models\Laporan;
use App\Models\User;
use Illuminate\Contracts\View\View;

class Dashboard extends AdminPage
{
    public function render(): View
    {
        $counts = Laporan::query()->selectRaw('status, count(*) as jumlah')->groupBy('status')->pluck('jumlah', 'status');
        $stats = array_map(fn ($status) => (int) ($counts[$status] ?? 0), array_combine(array_keys(Laporan::STATUSES), array_keys(Laporan::STATUSES)));
        $total = array_sum($stats);
        $citizens = User::where('role', 'warga')->count();
        $recent = Laporan::with('pelapor')->latest()->orderByDesc('id')->limit(5)->get();

        return view('livewire.admin.dashboard', compact('stats', 'total', 'citizens', 'recent'))->title('Dashboard Admin');
    }
}

<?php

namespace App\Livewire\Admin\Laporan;

use App\Livewire\AdminPage;
use App\Models\Laporan;
use Illuminate\Contracts\View\View;
use Illuminate\Validation\Rule;
use Livewire\WithPagination;

class Index extends AdminPage
{
    use WithPagination;

    public string $search = '';

    public string $status = '';

    public string $sort = 'newest';

    public function updatedSearch(): void
    {
        $this->resetPage();
    }

    public function updatedStatus(): void
    {
        $this->resetPage();
    }

    public function updatedSort(): void
    {
        $this->resetPage();
    }

    public function render(): View
    {
        $this->validate([
            'search' => 'max:255',
            'status' => ['nullable', Rule::in(array_keys(Laporan::STATUSES))],
            'sort' => ['required', Rule::in(['newest', 'oldest'])],
        ]);
        $direction = $this->sort === 'oldest' ? 'asc' : 'desc';
        $laporans = Laporan::with('pelapor')
            ->when($this->search !== '', fn ($q) => $q->where(function ($q): void {
                $q->where('judul', 'like', '%'.$this->search.'%')
                    ->orWhereHas('pelapor', fn ($q) => $q->where('name', 'like', '%'.$this->search.'%'));
            }))
            ->when($this->status !== '', fn ($q) => $q->where('status', $this->status))
            ->orderBy('created_at', $direction)->orderBy('id', $direction)->paginate(10);

        return view('livewire.admin.laporan.index', compact('laporans'))->title('Semua Pengaduan');
    }
}

<?php

namespace App\Livewire\Admin\Laporan;

use App\Livewire\AdminPage;
use App\Models\Laporan;
use App\Services\ComplaintManager;
use Illuminate\Contracts\View\View;
use Livewire\Attributes\Locked;

class Show extends AdminPage
{
    #[Locked]
    public int $laporanId;

    public function mount(Laporan $laporan): void
    {
        $this->authorize('changeStatus', $laporan);
        $this->laporanId = $laporan->id;
    }

    public function changeStatus(string $status): void
    {
        app(ComplaintManager::class)->transition($this->laporanId, $status);
        session()->flash('success', 'Status pengaduan berhasil diperbarui.');
    }

    public function render(): View
    {
        $laporan = Laporan::with('pelapor')->findOrFail($this->laporanId);
        $this->authorize('changeStatus', $laporan);

        return view('livewire.admin.laporan.show', compact('laporan'))->title('Tinjau Pengaduan');
    }
}

database/seeders/AdminSeeder.php
<?php

namespace Database\Seeders;

use App\Models\User;
use Illuminate\Database\Seeder;
use Illuminate\Support\Facades\Validator;
use Illuminate\Validation\Rules\Password;

class AdminSeeder extends Seeder
{
    public function run(): void
    {
        $email = config('ereport.admin_email');
        $password = config('ereport.admin_password');
        Validator::make(
            compact('email', 'password'),
            ['email' => 'required|email', 'password' => ['required', 'string', Password::min(12)->letters()->mixedCase()->numbers()->symbols()]],
            ['password.required' => 'Isi ADMIN_PASSWORD pada .env sebelum menjalankan seeder admin.']
        )->validate();

        if (User::where('email', $email)->exists()) {
            $this->command?->warn('Email sudah terdaftar. Seeder tidak mengubah akun atau password yang ada.');

            return;
        }

        $user = new User([
            'name' => 'Administrator E-Report',
            'email' => $email,
            'password' => $password,
            'no_whatsapp' => '',
            'alamat' => '',
        ]);
        $user->role = 'admin';
        $user->save();
    }
}
